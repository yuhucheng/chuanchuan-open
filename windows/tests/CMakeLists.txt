cmake_minimum_required(VERSION 3.14)
project(share_hub_windows_native_tests LANGUAGES CXX)
enable_testing()
add_executable(platform_tests platform_tests.cpp ../platform/discovery_model.cpp)
target_include_directories(platform_tests PRIVATE ../platform)
target_compile_features(platform_tests PRIVATE cxx_std_17)
target_compile_options(platform_tests PRIVATE /W4 /WX /utf-8 /EHsc)
add_test(NAME platform_tests COMMAND platform_tests)
add_executable(device_preferences_tests device_preferences_tests.cpp ../platform/device_preferences.cpp ../platform/discovery_model.cpp)

target_compile_definitions(device_preferences_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
